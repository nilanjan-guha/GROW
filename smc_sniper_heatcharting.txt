f_minutes(string tf) =>
    float result = na
    string t = str.upper(tf)
    if t == "D" or t == "1D"
        result := 1440
    else if t == "W" or t == "1W"
        result := 10080
    else if t == "M" or t == "1M"
        result := 43200
    else
        float n = str.tonumber(tf)
        if not na(n)
            result := n
    result

f_period(string tf) =>
    int result = na
    string t = str.upper(tf)
    int yy = year(time, tz)
    int mm = month(time, tz)
    int dd = dayofmonth(time, tz)
    if t == "M" or t == "1M"
        result := yy * 100 + mm
    else if t == "W" or t == "1W"
        int dow = dayofweek(time, tz)
        int daysFromMonday = (dow + 5) % 7
        int weekStart = time - daysFromMonday * 86400000
        result := int(math.floor(weekStart / 604800000))
    else if t == "D" or t == "1D"
        result := yy * 10000 + mm * 100 + dd
    else
        float mins = f_minutes(tf)
        if not na(mins) and mins > 0
            int minuteOfDay = hour(time, tz) * 60 + minute(time, tz)
            int bucket = int(math.floor(minuteOfDay / mins))
            result := (yy * 10000 + mm * 100 + dd) * 10000 + bucket
    result

// SWING AGGREGATION ENGINE

f_swings(string tf, int highStrength, int lowStrength, bool enabled) =>
    var float aggH = na
    var float aggL = na
    var int aggX = na
    var int lastPeriod = na

    var array<float> highs = array.new_float()
    var array<float> lows = array.new_float()
    var array<int> indexes = array.new_int()

    float newHigh = na
    float newLow = na
    int newHighX = na
    int newLowX = na

    if enabled
        int p = f_period(tf)
        if not na(p)
            if na(lastPeriod)
                aggH := high
                aggL := low
                aggX := bar_index
                lastPeriod := p
            else if p != lastPeriod
                array.push(highs, aggH)
                array.push(lows, aggL)
                array.push(indexes, aggX)

                int n = array.size(highs)
                if n >= highStrength * 2 + 1
                    int pivot = n - 1 - highStrength
                    float candidate = array.get(highs, pivot)
                    bool found = true
                    for i = 1 to highStrength
                        if candidate <= array.get(highs, pivot - i) or candidate <= array.get(highs, pivot + i)
                            found := false
                    if found
                        newHigh := candidate
                        newHighX := array.get(indexes, pivot)

                if n >= lowStrength * 2 + 1
                    int pivot = n - 1 - lowStrength
                    float candidate = array.get(lows, pivot)
                    bool found = true
                    for i = 1 to lowStrength
                        if candidate >= array.get(lows, pivot - i) or candidate >= array.get(lows, pivot + i)
                            found := false
                    if found
                        newLow := candidate
                        newLowX := array.get(indexes, pivot)

                int keep = math.max(highStrength, lowStrength) * 2 + 3
                while array.size(highs) > keep
                    array.shift(highs)
                    array.shift(lows)
                    array.shift(indexes)

                aggH := high
                aggL := low
                aggX := bar_index
                lastPeriod := p
            else
                aggH := math.max(aggH, high)
                aggL := math.min(aggL, low)

    [newHigh, newHighX, newLow, newLowX]
